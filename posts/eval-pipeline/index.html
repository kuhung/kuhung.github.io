### 函数计算
def reduce_mem_usage(df):
    # 计算初始内存使用
    start_mem = df.memory_usage().sum() / 1024**2
    print(f'Memory usage of dataframe is {start_mem:.2f} MB')
    
    # 循环遍历df中的所有列。
    for col in df.columns:
        col_type = df[col].dtype
        
        if col_type != object:
            c_min = df[col].min()
            c_max = df[col].max()
            
            if str(col_type)[:3] == 'int':
                # 如果是整数，代码会检查最小和最大值，选择一个更小的整数类型来存储数据。
                # 例如，如果所有值都能用np.int8表示，就将数据类型更改为np.int8。
                if c_min > np.iinfo(np.int8).min and c_max < np.iinfo(np.int8).max:
                    df[col] = df[col].astype(np.int8)
                elif c_min > np.iinfo(np.int16).min and c_max < np.iinfo(np.int16).max:
                    df[col] = df[col].astype(np.int16)
                elif c_min > np.iinfo(np.int32).min and c_max < np.iinfo(np.int32).max:
                    df[col] = df[col].astype(np.int32)
                elif c_min > np.iinfo(np.int64).min and c_max < np.iinfo(np.int64).max:
                    df[col] = df[col].astype(np.int64)
            
            else:
                # 对于浮点数，选择一个更小的浮点数类型。
                if c_min > np.finfo(np.float16).min and c_max < np.finfo(np.float16).max:
                    df[col] = df[col].astype(np.float16)
                elif c_min > np.finfo(np.float32).min and c_max < np.finfo(np.float32).max:
                    df[col] = df[col].astype(np.float32)
                else:
                    df[col] = df[col].astype(np.float32)
    
    # 计算最终内存
    end_mem = df.memory_usage().sum() / 1024**2
    print(f'Memory usage after optimization is: {end_mem:.2f} MB')
    
    # 计算内存减少的百分比
    decrease = 100 * (start_mem - end_mem) / start_mem
    print(f'Decreased by {decrease:.2f}%')
    
    return df

各个损失函数之间的比较如下：

  1. CategoricalCrossentropy：计算标签和预测之间的交叉熵损失。
  2. CategoricalFocalCrossentropy：计算 alpha 平衡的焦点交叉熵损失。
  3. CategoricalHinge：计算标签和预测之间的多分类铰链损失。
  4. CosineSimilarity：计算标签和预测之间的余弦相似度。
  5. Hinge：计算标签和预测之间的铰链损失。
  6. Huber：计算标签和预测之间的 Huber 损失。
  7. KLDivergence：计算标签和预测之间的 Kullback-Leibler 散度损失。
  8. LogCosh：计算预测误差的双曲余弦对数。
  9. Loss：损失函数基类。
  10. MeanAbsoluteError：计算标签和预测之间绝对差的平均值。
  11. MeanAbsolutePercentageError：计算标签和预测之间平均绝对百分比误差。
  12. MeanSquaredError：计算标签和预测之间误差平方的平均值。
  13. MeanSquaredLogarithmicError：计算标签和预测之间均方对数误差。
  14. Poisson：计算标签和预测之间的泊松损失。
  15. Reduction：损失函数减小的类型。
  16. SparseCategoricalCrossentropy：计算标签和预测之间的稀疏交叉熵损失。
  17. SquaredHinge：计算标签和预测之间的平方铰链损失。

这些不同的损失函数适用于不同的问题和模型类型。你可以根据你的任务需求和数据特点选择合适的损失函数来训练模型。

  • https://dvc.org/

  1. Wandb（Weights & Biases）：Wandb是一个强大的实验记录和可视化工具包，它可以帮助你跟踪和组织你的机器学习实验。它提供了用于记录和比较指标、可视化模型训练过程、保存和共享模型等功能。

  2. MLflow：MLflow是一个开源的机器学习生命周期管理平台，提供了对实验记录和管理的支持。它可以帮助你跟踪各个实验运行的参数、指标和模型，以便重现实验和进行模型版本管理。

    1. 参考文章：https://www.atyun.com/56891.html

  3. Neptune：Neptune是一个具有实验管理和版本控制功能的机器学习元数据中心。它可以追踪实验参数、指标、模型、原始数据和图形，并提供协作和共享功能。
