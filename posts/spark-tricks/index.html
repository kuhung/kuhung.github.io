pyspark 自定义 python

# https://conda.github.io/conda-pack/spark.html
# note: script.py 中需要完整的 spark session 初始化

PYSPARK_PYTHON=./environment/bin/python \
spark-submit \
--conf spark.yarn.appMasterEnv.PYSPARK_PYTHON=./environment/bin/python \
--master yarn \
--deploy-mode cluster \
--archives environment.tar.gz#environment \
script.py

jupyter 中使用 pyspark 交互式开发

# https://queirozf.com/entries/jupyter-kernels-how-to-add-change-remove#change-kernel-name
conda create --name pyspark python=3.7 xgboost scikit-learn jupyter pandas py4j findspark
conda activate pyspark
ipython kernel install --name "pyspark" --user

#remove if config failed
conda env remove --name pyspark
jupyter kernelspec remove pyspark
